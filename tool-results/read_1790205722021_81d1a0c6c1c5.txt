    65→
    66→interface PendingComment {
    67→  id: number;
    68→  sectionKey: string;
    69→  comment: string;
    70→}
    71→
    72→let pendingSeq = 0;
    73→
    74→const REVIEW_ACTION_META: Record<string, { label: string; cls: string }> = {
    75→  START_REVIEW: { label: "Review started", cls: "border-amber-300 bg-amber-50 text-amber-800" },
    76→  COMMENT: { label: "Comments added", cls: "border-violet-300 bg-violet-50 text-violet-800" },
    77→  REVISION_REQUESTED: { label: "Revision requested", cls: "border-orange-300 bg-orange-50 text-orange-800" },
    78→  APPROVED: { label: "Approved", cls: "border-emerald-300 bg-emerald-50 text-emerald-800" },
    79→  RATED: { label: "Rated", cls: "border-teal-300 bg-teal-50 text-teal-800" },
    80→  ARCHIVED: { label: "Archived", cls: "border-zinc-300 bg-zinc-100 text-zinc-700" },
    81→};
    82→
    83→function reviewActionBadge(action: string) {
    84→  const meta = REVIEW_ACTION_META[action] ?? { label: action.replace(/_/g, " "), cls: "border-slate-300 bg-slate-100 text-slate-700" };
    85→  return <span className={cn("inline-flex items-center rounded-full border px-2 py-0.5 text-[11px] font-semibold", meta.cls)}>{meta.label}</span>;
    86→}
    87→
    88→export default function MdrrmoReview({
    89→  id,
    90→  onBack,
    91→  onChanged,
    92→}: {
    93→  id: string;
    94→  onBack: () => void;
    95→  onChanged?: () => void;
    96→}) {
    97→  const { toast } = useToast();
    98→  const { data: detail, loading, error, reload } = useLoad<AdminSubmissionDetail>(() => api.adminSubmission(id), id);
    99→  const [tab, setTab] = useState("submission");
   100→  const [drafts, setDrafts] = useState<Record<string, string>>({});
   101→  const [pending, setPending] = useState<PendingComment[]>([]);
   102→  const [busy, setBusy] = useState(false);
   103→
   104→  // dialog state
   105→  const [revisionOpen, setRevisionOpen] = useState(false);
   106→  const [revisionOverall, setRevisionOverall] = useState("");
   107→  const [approveOpen, setApproveOpen] = useState(false);
   108→  const [approveOverall, setApproveOverall] = useState("");
   109→  const [archiveOpen, setArchiveOpen] = useState(false);
   110→  const [finalizeOpen, setFinalizeOpen] = useState(false);
   111→  const [finalizeName, setFinalizeName] = useState("");
   112→  const [finalizePosition, setFinalizePosition] = useState("");
   113→  const [finalizing, setFinalizing] = useState(false);
   114→  const [finalizeResult, setFinalizeResult] = useState<Awaited<ReturnType<typeof api.adminFinalize>> | null>(null);
   115→
   116→  // evaluation form state
   117→  const [scores, setScores] = useState<Record<string, number>>({});
   118→  const [remarks, setRemarks] = useState("");
   119→
   120→  const commentsBySection = useMemo(() => {
   121→    const map = new Map<string, CommentItem[]>();
   122→    for (const c of detail?.comments ?? []) {
   123→      const list = map.get(c.sectionKey) ?? [];
   124→      list.push(c);
   125→      map.set(c.sectionKey, list);
   126→    }
   127→    return map;
   128→  }, [detail?.comments]);
   129→  const filesBySection = useMemo(() => {
   130→    const map = new Map<string, FileMeta[]>();
   131→    for (const f of detail?.files ?? []) {
   132→      const list = map.get(f.sectionKey) ?? [];
   133→      list.push(f);
   134→      map.set(f.sectionKey, list);
   135→    }
   136→    return map;
   137→  }, [detail?.files]);
   138→
   139→  useEffect(() => {
   140→    if (!detail?.rating) return;
   141→    const s: Record<string, number> = {};
   142→    for (const c of detail.rating.scores) s[c.key] = c.score;
   143→    setScores(s);
   144→    setRemarks(detail.rating.remarks ?? "");
   145→  }, [detail]);
   146→
   147→  const st = detail?.submission.status;
   148→  const canStartReview = st === "SUBMITTED" || st === "RESUBMITTED";
   149→  const reviewable = !!st && ["SUBMITTED", "RESUBMITTED", "UNDER_REVIEW", "NEEDS_REVISION"].includes(st);
   150→  const canRequestRevision = !!st && ["SUBMITTED", "RESUBMITTED", "UNDER_REVIEW"].includes(st);
   151→  const canApprove = reviewable;
   152→  const canFinalize = st === "APPROVED";
   153→  const canRegenerate = st === "READY_FOR_DOWNLOAD";
   154→  const canArchive = !!st && ["APPROVED", "READY_FOR_DOWNLOAD", "DOWNLOADED"].includes(st);
   155→  const hasRating = !!detail?.rating;
   156→
   157→  const openFinalize = async () => {
   158→    try {
   159→      const res = await api.adminSettings();
   160→      setFinalizeName(res.settings.signatoryName);
   161→      setFinalizePosition(res.settings.signatoryPosition);
   162→    } catch {
   163→      // fall back to blank inputs
   164→    }
   165→    setFinalizeOpen(true);
   166→  };
   167→
   168→  const runReviewAction = async (
   169→    payload: Record<string, unknown>,
   170→    successTitle: string,
   171→    opts?: { clearPending?: boolean; closeDialog?: () => void }
   172→  ) => {
   173→    setBusy(true);
   174→    try {
   175→      await api.adminReview(id, payload);
   176→      toast({ title: successTitle });
   177→      if (opts?.clearPending) {
   178→        setPending([]);
   179→        setDrafts({});
   180→      }
   181→      opts?.closeDialog?.();
   182→      reload();
   183→      onChanged?.();
   184→    } catch (e) {
   185→      toast({
   186→        title: "Action failed",
   187→        description: e instanceof Error ? e.message : "Please try again.",
   188→        variant: "destructive",
   189→      });
   190→    } finally {
   191→      setBusy(false);
   192→    }
   193→  };
   194→
   195→  const doFinalize = async () => {
   196→    setFinalizing(true);
   197→    try {
   198→      const res = await api.adminFinalize(id, finalizeName, finalizePosition);
   199→      setFinalizeResult(res);
   200→      setFinalizeOpen(false);
   201→      reload();
   202→      onChanged?.();
   203→    } catch (e) {
   204→      toast({
   205→        title: "Finalization failed",
   206→        description: e instanceof Error ? e.message : "Please try again.",
   207→        variant: "destructive",
   208→      });
   209→    } finally {
   210→      setFinalizing(false);
   211→    }
   212→  };
   213→
   214→  const addPending = (sectionKey: string) => {
   215→    const text = (drafts[sectionKey] ?? "").trim();
   216→    if (!text) return;
   217→    setPending((p) => [...p, { id: ++pendingSeq, sectionKey, comment: text }]);
   218→    setDrafts((d) => ({ ...d, [sectionKey]: "" }));
   219→  };
   220→
   221→  if (loading && !detail) {
   222→    return (
   223→      <div className="space-y-4">
   224→        <Button variant="ghost" size="sm" onClick={onBack}>
   225→          <ArrowLeft className="h-4 w-4" /> Back
   226→        </Button>
   227→        <Skeleton className="h-36 rounded-xl" />
   228→        <Skeleton className="h-12 w-72 rounded-lg" />
   229→        <Skeleton className="h-96 rounded-xl" />
   230→      </div>
   231→    );
   232→  }
   233→  if (error || !detail) {
   234→    return (
   235→      <div className="space-y-4">
   236→        <Button variant="ghost" size="sm" onClick={onBack}>
   237→          <ArrowLeft className="h-4 w-4" /> Back
   238→        </Button>
   239→        <ErrorAlert message={error ?? "Submission not found"} onRetry={reload} />
   240→      </div>
   241→    );
   242→  }
   243→
   244→  const lang = detail.submission.templateLang;
   245→
   246→  return (
   247→    <div className="space-y-4 pb-4">
   248→      <Button variant="ghost" size="sm" onClick={onBack} className="-ml-2">
   249→        <ArrowLeft className="h-4 w-4" /> Back
   250→      </Button>
   251→
   252→      <div className={cn("grid gap-4", detail.document ? "xl:grid-cols-3" : "")}>
   253→        {/* Header card */}
   254→        <Card className={cn(detail.document ? "xl:col-span-2" : "")}>
   255→          <CardContent className="space-y-4">
   256→            <div className="flex flex-wrap items-start justify-between gap-4">
   257→              <div className="min-w-0 space-y-1.5">
   258→                <div className="flex flex-wrap items-center gap-2">
   259→                  <h1 className="text-lg font-semibold tracking-tight">Barangay {detail.barangay.name}</h1>
   260→                  <code className="rounded-md bg-muted px-1.5 py-0.5 font-mono text-xs text-muted-foreground">{detail.barangay.code}</code>
   261→                  <StatusBadge status={detail.submission.status} />
   262→                </div>
   263→                <p className="text-sm text-muted-foreground">
   264→                  {detail.barangay.captain ? `Punong Barangay ${detail.barangay.captain} · ` : ""}BDRRMP {detail.submission.year}
   265→                </p>
   266→                <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-muted-foreground">
   267→                  <span>
   268→                    Version <span className="font-mono font-semibold text-foreground">v{detail.submission.version}</span>
   269→                  </span>
   270→                  <span>
   271→                    Template{" "}
   272→                    <span className="rounded border border-border bg-muted px-1 py-0.5 font-semibold text-foreground">{lang === "TL" ? "TL" : "EN"}</span>
   273→                  </span>
   274→                  <span>Submitted {formatDateTime(detail.submission.submittedAt)}</span>
   275→                </div>
   276→                <div className="flex max-w-xs items-center gap-2 pt-1">
   277→                  <Progress value={detail.submission.progress} className="h-2 flex-1" />
   278→                  <span className="text-xs tabular-nums text-muted-foreground">{detail.submission.progress}%</span>
   279→                </div>
   280→              </div>
   281→
   282→              {/* Contextual actions */}
   283→              <div className="flex flex-wrap items-center gap-2">
   284→                {canStartReview && (
   285→                  <Button disabled={busy} onClick={() => void runReviewAction({ action: "start" }, "Review started")}>
   286→                    <ClipboardList className="h-4 w-4" /> Start Review
   287→                  </Button>
   288→                )}
   289→                {st === "UNDER_REVIEW" && (
   290→                  <Button
   291→                    variant="outline"
   292→                    onClick={() => {
   293→                      setTab("submission");
   294→                      toast({ title: "Add comments per section", description: "Write a comment under any section below, then submit them from the action bar." });
   295→                    }}
   296→                  >
   297→                    <MessageSquare className="h-4 w-4" /> Add Comments
   298→                  </Button>
   299→                )}
   300→                {canRequestRevision && (
   301→                  <Button
   302→                    variant="outline"
   303→                    disabled={pending.length === 0 || busy}
   304→                    className="border-amber-300 text-amber-800 hover:bg-amber-50 dark:text-amber-400 dark:hover:bg-amber-950/40"
   305→                    onClick={() => setRevisionOpen(true)}
   306→                  >
   307→                    <Undo2 className="h-4 w-4" /> Request Revision
   308→                  </Button>
   309→                )}
   310→                {canApprove &&
   311→                  (hasRating ? (
   312→                    <Button
   313→                      disabled={busy}
   314→                      className="bg-emerald-600 text-white hover:bg-emerald-700 dark:bg-emerald-500 dark:hover:bg-emerald-600"
   315→                      onClick={() => setApproveOpen(true)}
   316→                    >
   317→                      <CheckCircle2 className="h-4 w-4" /> Approve
   318→                    </Button>
   319→                  ) : (
   320→                    <HintButton hint="Complete the evaluation first (Evaluation tab)">
   321→                      <Button disabled className="bg-emerald-600 text-white hover:bg-emerald-700 dark:bg-emerald-500">
   322→                        <CheckCircle2 className="h-4 w-4" /> Approve
   323→                      </Button>
   324→                    </HintButton>
   325→                  ))}
   326→                {canFinalize && (
   327→                  <Button onClick={() => void openFinalize()}>
   328→                    <FileSignature className="h-4 w-4" /> Finalize Document
   329→                  </Button>
   330→                )}
   331→                {canRegenerate && (
   332→                  <Button variant="outline" onClick={() => void openFinalize()}>
   333→                    <FileSignature className="h-4 w-4" /> Regenerate Final Document
   334→                  </Button>
   335→                )}
   336→                {canArchive && (
   337→                  <Button variant="ghost" disabled={busy} onClick={() => setArchiveOpen(true)}>
   338→                    <Archive className="h-4 w-4" /> Archive
   339→                  </Button>
   340→                )}
   341→              </div>
   342→            </div>
   343→
   344→            {st === "NEEDS_REVISION" && (
   345→              <Alert className="border-amber-300 bg-amber-50 text-amber-900 dark:border-amber-800 dark:bg-amber-950/40 dark:text-amber-200">
   346→                <Clock className="h-4 w-4" />
   347→                <AlertTitle>Waiting for barangay to resubmit</AlertTitle>
   348→                <AlertDescription className="text-amber-800 dark:text-amber-300">
   349→                  A revision request was sent to the barangay. You can still review the current version and approve it directly if the
   350→                  submitted revision is already satisfactory.
   351→                </AlertDescription>
   352→              </Alert>
   353→            )}
   354→            {!reviewable && st !== undefined && ["NOT_STARTED", "DRAFT", "READY_FOR_SUBMISSION"].includes(st) && (
   355→              <Alert>
   356→                <AlertTriangle className="h-4 w-4" />
   357→                <AlertTitle>Not yet submitted</AlertTitle>
   358→                <AlertDescription>
   359→                  This barangay has not submitted its BDRRMP yet — the content below is a read-only view of their draft.
   360→                </AlertDescription>
   361→              </Alert>
   362→            )}
   363→          </CardContent>
   364→        </Card>
   365→
   366→        {/* Final document card */}
   367→        {detail.document && <DocumentCard document={detail.document} />}
   368→      </div>
   369→
   370→      <Tabs value={tab} onValueChange={setTab}>
   371→        <TabsList>
   372→          <TabsTrigger value="submission">Submission</TabsTrigger>
   373→          <TabsTrigger value="evaluation">Evaluation</TabsTrigger>
   374→          <TabsTrigger value="history">
   375→            <History className="h-4 w-4" /> History
   376→          </TabsTrigger>
   377→        </TabsList>
   378→
   379→        {/* ---------------- Submission tab ---------------- */}
   380→        <TabsContent value="submission" className="space-y-3">
   381→          <Accordion type="single" collapsible defaultValue={detail.sections[0]?.key} className="space-y-3">
   382→            {detail.sections.map((section, i) => (
   383→              <SectionAccordion
   384→                key={section.key}
   385→                index={i + 1}
   386→                section={section}
   387→                lang={lang}
   388→                values={detail.submission.values}
   389→                files={filesBySection.get(section.key) ?? []}
   390→                comments={commentsBySection.get(section.key) ?? []}
   391→                pendingComments={pending.filter((p) => p.sectionKey === section.key)}
   392→                draft={drafts[section.key] ?? ""}
   393→                onDraftChange={(v) => setDrafts((d) => ({ ...d, [section.key]: v }))}
   394→                onAddPending={() => addPending(section.key)}
   395→                onRemovePending={(pid) => setPending((p) => p.filter((x) => x.id !== pid))}
   396→                reviewable={reviewable}
   397→                busy={busy}
   398→              />
   399→            ))}
   400→          </Accordion>
   401→        </TabsContent>
   402→
   403→        {/* ---------------- Evaluation tab ---------------- */}
   404→        <TabsContent value="evaluation">
   405→          <Card className="max-w-2xl">
   406→            <CardHeader>
   407→              <CardTitle className="text-base">MDRRMO EVALUATION</CardTitle>
   408→              <CardDescription>
   409→                Quality Assurance Team rating for Barangay {detail.barangay.name}&apos;s BDRRMP {detail.submission.year} (v
   410→                {detail.submission.version})
   411→              </CardDescription>
   412→            </CardHeader>
   413→            <CardContent className="space-y-4">
   414→              {detail.criteria.length === 0 && <p className="text-sm text-muted-foreground">No rating criteria configured (see Settings).</p>}
   415→              {detail.criteria.map((c) => (
   416→                <div key={c.key} className="flex items-center gap-3">
   417→                  <Label htmlFor={`score-${c.key}`} className="flex-1 text-sm font-normal">
   418→                    {c.name}
   419→                  </Label>
   420→                  <Input
   421→                    id={`score-${c.key}`}
   422→                    type="number"
   423→                    min={0}
   424→                    max={c.maxScore}
   425→                    inputMode="numeric"
   426→                    className="w-20 text-right tabular-nums"
   427→                    value={scores[c.key] ?? ""}
   428→                    onChange={(e) => {
   429→                      const v = e.target.value === "" ? undefined : Number(e.target.value);
   430→                      setScores((s) => {
   431→                        const next = { ...s };
   432→                        if (v === undefined || Number.isNaN(v)) delete next[c.key];
   433→                        else next[c.key] = v;
   434→                        return next;
   435→                      });
   436→                    }}
   437→                    aria-label={`Score for ${c.name}`}
   438→                  />
   439→                  <span className="w-12 text-sm text-muted-foreground">/ {c.maxScore}</span>
   440→                </div>
   441→              ))}
   442→              <div className="flex items-center justify-between border-t pt-3">
   443→                <span className="text-sm font-medium">Total</span>
   444→                <span className="text-lg font-semibold tabular-nums">
   445→                  {detail.criteria.reduce((a, c) => a + (scores[c.key] ?? 0), 0)} / {detail.criteria.reduce((a, c) => a + c.maxScore, 0)}
   446→                </span>
   447→              </div>
   448→              <div className="space-y-1.5">
   449→                <Label htmlFor="eval-remarks" className="text-sm font-normal">
   450→                  Remarks
   451→                </Label>
   452→                <Textarea id="eval-remarks" rows={3} value={remarks} onChange={(e) => setRemarks(e.target.value)} placeholder="Overall observations, strengths, recommendations..." />
   453→              </div>
   454→              <p className="text-xs text-muted-foreground">
   455→                Score guide: {detail.criteria[0]?.maxScore ?? 25} = Excellent, 0 = Missing
   456→              </p>
   457→              {detail.rating && (
   458→                <p className="text-xs text-muted-foreground">
   459→                  Last saved: {detail.rating.total}/{detail.rating.maxTotal}
   460→                  {detail.rating.ratedByName ? ` by ${detail.rating.ratedByName}` : ""} on {formatDateTime(detail.rating.createdAt)}
   461→                </p>
   462→              )}
   463→              <div>
   464→                <Button
   465→                  disabled={
   466→                    busy ||
   467→                    detail.criteria.length === 0 ||
   468→                    !detail.criteria.every((c) => scores[c.key] !== undefined && scores[c.key] >= 0 && scores[c.key] <= c.maxScore)
   469→                  }
   470→                  onClick={async () => {
   471→                    setBusy(true);
   472→                    try {
   473→                      await api.adminRating(id, scores, remarks);
   474→                      toast({ title: "Evaluation saved" });
   475→                      reload();
   476→                      onChanged?.();
   477→                    } catch (e) {
   478→                      toast({
   479→                        title: "Could not save evaluation",
   480→                        description: e instanceof Error ? e.message : "Please try again.",
   481→                        variant: "destructive",
   482→                      });
   483→                    } finally {
   484→                      setBusy(false);
   485→                    }
   486→                  }}
   487→                >
   488→                  Save Evaluation
   489→                </Button>
   490→              </div>
   491→            </CardContent>
   492→          </Card>
   493→        </TabsContent>
   494→
   495→        {/* ---------------- History tab ---------------- */}
   496→        <TabsContent value="history" className="grid gap-4 lg:grid-cols-2">
   497→          <Card className="self-start">
   498→            <CardHeader>
   499→              <CardTitle className="text-base">Review Timeline</CardTitle>
   500→              <CardDescription>All MDRRMO actions on this submission</CardDescription>
   501→            </CardHeader>
   502→            <CardContent>
   503→              {detail.reviews.length === 0 && <p className="text-sm text-muted-foreground">No reviews recorded yet.</p>}
   504→              <ul className="space-y-4">
   505→                {detail.reviews
   506→                  .slice()
   507→                  .reverse()
   508→                  .map((r) => (
   509→                    <li key={r.id} className="relative border-l-2 border-primary/30 pl-4">
   510→                      <span className="absolute -left-[5px] top-1 h-2 w-2 rounded-full bg-primary" />
   511→                      <div className="flex flex-wrap items-center gap-2">
   512→                        {reviewActionBadge(r.action)}
   513→                        <span className="text-sm font-medium">{r.reviewerName}</span>
   514→                        <span className="font-mono text-xs text-muted-foreground">v{r.version}</span>

   821→// Wraps a (possibly disabled) button with a tooltip — disabled buttons swallow
   822→// pointer events, so the trigger must sit on a wrapping span.
   823→function HintButton({ hint, children }: { hint: string; children: ReactNode }) {
   824→  return (
   825→    <TooltipProvider delayDuration={100}>
   826→      <Tooltip>
   827→        <TooltipTrigger asChild>
   828→          <span tabIndex={0} aria-label={hint} className="inline-flex">
   829→            {children}
   830→          </span>
   831→        </TooltipTrigger>
   832→        <TooltipContent side="top">{hint}</TooltipContent>
   833→      </Tooltip>
   834→    </TooltipProvider>
   835→  );
   836→}
