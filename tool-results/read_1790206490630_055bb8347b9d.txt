     9→export interface DbFieldDef {

    11→  label: string; // UI label
    12→  type: DbFieldType;

    14→  nullable?: boolean; // accepts empty -> null

   452→  count: (args?: any) => Promise<number>;
   453→  findMany: (args?: any) => Promise<any[]>;
   454→  findUnique: (args: any) => Promise<any | null>;
   455→  create: (args: any) => Promise<any>;
   456→  update: (args: any) => Promise<any>;
   457→  delete: (args: any) => Promise<any>;
   458→}
   459→
   460→function getModel(def: DbTableDef): DynamicModel {
   461→  const models = db as unknown as Record<string, DynamicModel>;
   462→  const model = models[def.model];
   463→  if (!model) throw new Error(`Unknown Prisma model: ${def.model}`);
   464→  return model;
   465→}

   467→export interface TableSummary {
   468→  key: string;
   469→  label: string;
   470→  group: string;
   471→  desc: string;
   472→  count: number;
   473→  fields: DbFieldDef[];
   474→  idField: string;
   475→  orderBy: { field: string; dir: "asc" | "desc" };
   476→  searchFields: string[];
   477→}
   478→
   479→export async function listTables(): Promise<TableSummary[]> {
   480→  const out: TableSummary[] = [];
   481→  for (const def of DB_TABLES) {
   482→    const count = await getModel(def).count().catch(() => 0);
   483→    out.push({
   484→      key: def.key,
   485→      label: def.label,
   486→      group: def.group,
   487→      desc: def.desc,
   488→      count,
   489→      fields: def.fields,
   490→      idField: def.idField,
   491→      orderBy: def.orderBy,
   492→      searchFields: def.searchFields,
   493→    });
   494→  }
   495→  return out;
   496→}
   497→
   498→function buildSearchWhere(def: DbTableDef, search: string) {
   499→  if (!search) return undefined;
   500→  return { OR: def.searchFields.map((f) => ({ [f]: { contains: search } })) };
   501→}

   503→export async function listRows(
   504→  tableKey: string,
   505→  opts: { page: number; pageSize: number; search: string }
   506→): Promise<{ rows: Record<string, unknown>[]; total: number; page: number; pageSize: number }> {
   507→  const def = getTableDef(tableKey);
   508→  if (!def) throw new Error("Unknown table");
   509→  const model = getModel(def);
   510→  const page = Math.max(1, opts.page);
   511→  const pageSize = Math.min(100, Math.max(5, opts.pageSize));
   512→  const where = buildSearchWhere(def, opts.search);
   513→  const [rows, total] = await Promise.all([
   514→    model.findMany({
   515→      where,
   516→      orderBy: { [def.orderBy.field]: def.orderBy.dir },
   517→      skip: (page - 1) * pageSize,
   518→      take: pageSize,
   519→    }),
   520→    model.count({ where }),
   521→  ]);
   522→  return { rows: rows as Record<string, unknown>[], total, page, pageSize };
   523→}

   525→// Coerce and validate a raw payload against the table's field definitions.
   526→// Returns { data } ready for Prisma, or throws with a readable message.
   527→export function coerceRowData(
   528→  def: DbTableDef,
   529→  body: Record<string, unknown>,
   530→  mode: "create" | "update"
   531→): Record<string, unknown> {
   532→  const data: Record<string, unknown> = {};
   533→  for (const field of def.fields) {
   534→    const present = Object.prototype.hasOwnProperty.call(body, field.name);
   535→    if (mode === "create") {
   536→      if (field.readonly) continue; // auto-generated
   537→      if (!present) {
   538→        if (field.required) throw new Error(`Field "${field.label}" is required.`);
   539→        continue; // optional fields fall back to Prisma defaults
   540→      }
   541→    } else {
   542→      if (!present || field.readonly) continue;
   543→    }
   544→    const raw = body[field.name];
   545→    if (raw === null || raw === "" || raw === undefined) {
   546→      if (field.type === "boolean") {
   547→        data[field.name] = false;
   548→        continue;
   549→      }
   550→      if (field.nullable) {
   551→        data[field.name] = null;
   552→        continue;
   553→      }
   554→      if (mode === "update") continue; // skip clearing non-nullable fields
   555→      throw new Error(`Field "${field.label}" cannot be empty.`);
   556→    }
   557→    switch (field.type) {
   558→      case "number": {
   559→        const n = Number(raw);
   560→        if (!Number.isFinite(n)) throw new Error(`Field "${field.label}" must be a number.`);
   561→        data[field.name] = n;
   562→        break;
   563→      }
   564→      case "boolean":
   565→        data[field.name] = raw === true || raw === "true" || raw === 1 || raw === "1";
   566→        break;
   567→      case "datetime": {
   568→        const d = new Date(String(raw));
   569→        if (isNaN(d.getTime())) throw new Error(`Field "${field.label}" must be a valid date/time.`);
   570→        data[field.name] = d;
   571→        break;
   572→      }
   573→      case "json": {
   574→        try {
   575→          const parsed = JSON.parse(String(raw));
   576→          data[field.name] = JSON.stringify(parsed);
   577→        } catch {
   578→          throw new Error(`Field "${field.label}" must contain valid JSON.`);
   579→        }
   580→        break;
   581→      }
   582→      default:
   583→        data[field.name] = String(raw);
   584→    }
   585→  }
   586→  if (Object.keys(data).length === 0) throw new Error("Nothing to save — no editable fields were provided.");
   587→  return data;
   588→}

   590→export async function createRow(
   591→  tableKey: string,
   592→  body: Record<string, unknown>,
   593→  actor: { name: string; ip?: string }
   594→): Promise<Record<string, unknown>> {
   595→  const def = getTableDef(tableKey);
   596→  if (!def) throw new Error("Unknown table");
   597→  const data = coerceRowData(def, body, "create");
   598→  try {
   599→    const row = (await getModel(def).create({ data })) as Record<string, unknown>;
   600→    await logAudit({
   601→      actorType: "ADMIN",
   602→      actorName: actor.name,
   603→      action: "DB_CREATE",
   604→      detail: `Created row in ${def.label} (${def.idField}=${String(row[def.idField])}) via Database Management`,
   605→      ip: actor.ip,
   606→    });
   607→    return row;
   608→  } catch (e) {
   609→    throw new Error(friendlyDbError(e, def.label, "create"));
   610→  }
   611→}

   613→export async function updateRow(
   614→  tableKey: string,
   615→  id: string,
   616→  body: Record<string, unknown>,
   617→  actor: { name: string; ip?: string }
   618→): Promise<Record<string, unknown>> {
   619→  const def = getTableDef(tableKey);
   620→  if (!def) throw new Error("Unknown table");
   621→  const model = getModel(def);
   622→  const existing = await model.findUnique({ where: { [def.idField]: id } });
   623→  if (!existing) throw new Error("Row not found.");
   624→  const data = coerceRowData(def, body, "update");
   625→  try {
   626→    const row = (await model.update({ where: { [def.idField]: id }, data })) as Record<string, unknown>;
   627→    await logAudit({
   628→      actorType: "ADMIN",
   629→      actorName: actor.name,
   630→      action: "DB_UPDATE",
   631→      detail: `Updated row in ${def.label} (${def.idField}=${id}) via Database Management`,
   632→      ip: actor.ip,
   633→    });
   634→    return row;
   635→  } catch (e) {
   636→    throw new Error(friendlyDbError(e, def.label, "update"));
   637→  }
   638→}

   640→export async function deleteRow(
   641→  tableKey: string,
   642→  id: string,
   643→  actor: { id: string; name: string; ip?: string }
   644→): Promise<void> {
   645→  const def = getTableDef(tableKey);
   646→  if (!def) throw new Error("Unknown table");
   647→  const model = getModel(def);
   648→  const existing = await model.findUnique({ where: { [def.idField]: id } });
   649→  if (!existing) throw new Error("Row not found.");
   650→
   651→  // Guardrails for the admin_users table
   652→  if (def.key === "admin_users") {
   653→    const row = existing as unknown as { id: string; username: string; role: string; active: boolean };
   654→    if (row.id === actor.id) throw new Error("You cannot delete your own account.");
   655→    if (row.username === "sysadmin") throw new Error("The default System Administrator account (sysadmin) cannot be deleted.");
   656→    if (row.role === "SYSTEM_ADMIN") {
   657→      const admins = await db.adminUser.findMany({ where: { role: "SYSTEM_ADMIN", active: true } });
   658→      if (admins.length <= 1) throw new Error("Cannot delete the last active System Administrator account.");
   659→    }
   660→  }
   661→
   662→  try {
   663→    await model.delete({ where: { [def.idField]: id } });
   664→    await logAudit({
   665→      actorType: "ADMIN",
   666→      actorName: actor.name,
   667→      action: "DB_DELETE",
   668→      detail: `Deleted row from ${def.label} (${def.idField}=${id}) via Database Management`,
   669→      ip: actor.ip,
   670→    });
   671→  } catch (e) {
   672→    throw new Error(friendlyDbError(e, def.label, "delete"));
   673→  }
   674→}
   675→
   676→function friendlyDbError(e: unknown, label: string, verb: string): string {
   677→  const msg = e instanceof Error ? e.message : String(e);
   678→  if (msg.includes("UNIQUE constraint failed")) {
   679→    return `A row with the same unique value already exists in ${label}.`;
   680→  }
   681→  if (msg.includes("FOREIGN KEY constraint failed")) {
   682→    return `Cannot ${verb} this ${label} row — other records reference it.`;
   683→  }
   684→  if (msg.includes("Required value missing")) {
   685→    return `A required field is missing for ${label}.`;
   686→  }
   687→  return `Could not ${verb} the ${label} row: ${msg.slice(0, 200)}`;
   688→}
