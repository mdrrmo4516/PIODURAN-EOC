    87→    87→
    88→    88→export default function MdrrmoReview({
    89→    89→  id,
    90→    90→  onBack,
    91→    91→  onChanged,
    92→    92→}: {
    93→    93→  id: string;
    94→    94→  onBack: () => void;
    95→    95→  onChanged?: () => void;
    96→    96→}) {
    97→    97→  const { toast } = useToast();
    98→    98→  const { data: detail, loading, error, reload } = useLoad<AdminSubmissionDetail>(() => api.adminSubmission(id), id);
    99→    99→  const [tab, setTab] = useState("submission");
   100→   100→  const [drafts, setDrafts] = useState<Record<string, string>>({});
   101→   101→  const [pending, setPending] = useState<PendingComment[]>([]);
   102→   102→  const [busy, setBusy] = useState(false);
   103→   103→
   104→   104→  // dialog state
   105→   105→  const [revisionOpen, setRevisionOpen] = useState(false);
   106→   106→  const [revisionOverall, setRevisionOverall] = useState("");
   107→   107→  const [approveOpen, setApproveOpen] = useState(false);
   108→   108→  const [approveOverall, setApproveOverall] = useState("");
   109→   109→  const [archiveOpen, setArchiveOpen] = useState(false);
   110→   110→  const [finalizeOpen, setFinalizeOpen] = useState(false);
   111→   111→  const [finalizeName, setFinalizeName] = useState("");
   112→   112→  const [finalizePosition, setFinalizePosition] = useState("");
   113→   113→  const [finalizing, setFinalizing] = useState(false);
   114→   114→  const [finalizeResult, setFinalizeResult] = useState<Awaited<ReturnType<typeof api.adminFinalize>> | null>(null);
   115→   115→
   116→   116→  // evaluation form state
   117→   117→  const [scores, setScores] = useState<Record<string, number>>({});

   123→   123→      const list = map.get(c.sectionKey) ?? [];
   124→   124→      list.push(c);
   125→   125→      map.set(c.sectionKey, list);
   126→   126→    }
   127→   127→    return map;
   128→   128→  }, [detail?.comments]);
   129→   129→  const filesBySection = useMemo(() => {
   130→   130→    const map = new Map<string, FileMeta[]>();
   131→   131→    for (const f of detail?.files ?? []) {
   132→   132→      const list = map.get(f.sectionKey) ?? [];
   133→   133→      list.push(f);
   134→   134→      map.set(f.sectionKey, list);
   135→   135→    }
   136→   136→    return map;
   137→   137→  }, [detail?.files]);
   138→   138→
   139→   139→  useEffect(() => {
   140→   140→    if (!detail?.rating) return;
   141→   141→    const s: Record<string, number> = {};

   143→   143→    setScores(s);
   144→   144→    setRemarks(detail.rating.remarks ?? "");
   145→   145→  }, [detail]);
   146→   146→
   147→   147→  const st = detail?.submission.status;
   148→   148→  const canStartReview = st === "SUBMITTED" || st === "RESUBMITTED";

   152→   152→  const canFinalize = st === "APPROVED";
   153→   153→  const canRegenerate = st === "READY_FOR_DOWNLOAD";
   154→   154→  const canArchive = !!st && ["APPROVED", "READY_FOR_DOWNLOAD", "DOWNLOADED"].includes(st);

   166→   166→  };

   169→   169→    payload: Record<string, unknown>,
   170→   170→    successTitle: string,
   171→   171→    opts?: { clearPending?: boolean; closeDialog?: () => void }

   174→   174→    try {
   175→   175→      await api.adminReview(id, payload);

   178→   178→        setPending([]);
   179→   179→        setDrafts({});
   180→   180→      }
   181→   181→      opts?.closeDialog?.();
   182→   182→      reload();
   183→   183→      onChanged?.();
   184→   184→    } catch (e) {
   185→   185→      toast({
   186→   186→        title: "Action failed",
   187→   187→        description: e instanceof Error ? e.message : "Please try again.",
   188→   188→        variant: "destructive",
   189→   189→      });
   190→   190→    } finally {
   191→   191→      setBusy(false);
   192→   192→    }
   193→   193→  };
   194→   194→
   195→   195→  const doFinalize = async () => {
   196→   196→    setFinalizing(true);
   197→   197→    try {
   198→   198→      const res = await api.adminFinalize(id, finalizeName, finalizePosition);
   199→   199→      setFinalizeResult(res);
   200→   200→      setFinalizeOpen(false);
   201→   201→      reload();
   202→   202→      onChanged?.();
   203→   203→    } catch (e) {
   204→   204→      toast({
   205→   205→        title: "Finalization failed",
   206→   206→        description: e instanceof Error ? e.message : "Please try again.",
   207→   207→        variant: "destructive",
   208→   208→      });
   209→   209→    } finally {
   210→   210→      setFinalizing(false);
   211→   211→    }
   212→   212→  };
   213→   213→
   214→   214→  const addPending = (sectionKey: string) => {
   215→   215→    const text = (drafts[sectionKey] ?? "").trim();
   216→   216→    if (!text) return;
   217→   217→    setPending((p) => [...p, { id: ++pendingSeq, sectionKey, comment: text }]);
   218→   218→    setDrafts((d) => ({ ...d, [sectionKey]: "" }));
   219→   219→  };
   220→   220→
   221→   221→  if (loading && !detail) {
   222→   222→    return (
   223→   223→      <div className="space-y-4">
   224→   224→        <Button variant="ghost" size="sm" onClick={onBack}>
   225→   225→          <ArrowLeft className="h-4 w-4" /> Back
   226→   226→        </Button>
   227→   227→        <Skeleton className="h-36 rounded-xl" />
   228→   228→        <Skeleton className="h-12 w-72 rounded-lg" />
   229→   229→        <Skeleton className="h-96 rounded-xl" />
   230→   230→      </div>
   231→   231→    );
   232→   232→  }
   233→   233→  if (error || !detail) {
   234→   234→    return (
   235→   235→      <div className="space-y-4">
   236→   236→        <Button variant="ghost" size="sm" onClick={onBack}>
   237→   237→          <ArrowLeft className="h-4 w-4" /> Back
   238→   238→        </Button>
   239→   239→        <ErrorAlert message={error ?? "Submission not found"} onRetry={reload} />
   240→   240→      </div>
   241→   241→    );
   242→   242→  }
   243→   243→
   244→   244→  const lang = detail.submission.templateLang;
   245→   245→
   246→   246→  return (
   247→   247→    <div className="space-y-4 pb-4">
   248→   248→      <Button variant="ghost" size="sm" onClick={onBack} className="-ml-2">
   249→   249→        <ArrowLeft className="h-4 w-4" /> Back
   250→   250→      </Button>
   251→   251→
   252→   252→      <div className={cn("grid gap-4", detail.document ? "xl:grid-cols-3" : "")}>
   253→   253→        {/* Header card */}
   254→   254→        <Card className={cn(detail.document ? "xl:col-span-2" : "")}>
   255→   255→          <CardContent className="space-y-4">
   256→   256→            <div className="flex flex-wrap items-start justify-between gap-4">
   257→   257→              <div className="min-w-0 space-y-1.5">
   258→   258→                <div className="flex flex-wrap items-center gap-2">
   259→   259→                  <h1 className="text-lg font-semibold tracking-tight">Barangay {detail.barangay.name}</h1>
   260→   260→                  <code className="rounded-md bg-muted px-1.5 py-0.5 font-mono text-xs text-muted-foreground">{detail.barangay.code}</code>
   261→   261→                  <StatusBadge status={detail.submission.status} />
   262→   262→                </div>
   263→   263→                <p className="text-sm text-muted-foreground">
   264→   264→                  {detail.barangay.captain ? `Punong Barangay ${detail.barangay.captain} · ` : ""}BDRRMP {detail.submission.year}
   265→   265→                </p>
   266→   266→                <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-muted-foreground">
   267→   267→                  <span>
   268→   268→                    Version <span className="font-mono font-semibold text-foreground">v{detail.submission.version}</span>
   269→   269→                  </span>
   270→   270→                  <span>
   271→   271→                    Template{" "}
   272→   272→                    <span className="rounded border border-border bg-muted px-1 py-0.5 font-semibold text-foreground">{lang === "TL" ? "TL" : "EN"}</span>
   273→   273→                  </span>
   274→   274→                  <span>Submitted {formatDateTime(detail.submission.submittedAt)}</span>
   275→   275→                </div>
   276→   276→                <div className="flex max-w-xs items-center gap-2 pt-1">
   277→   277→                  <Progress value={detail.submission.progress} className="h-2 flex-1" />
   278→   278→                  <span className="text-xs tabular-nums text-muted-foreground">{detail.submission.progress}%</span>
   279→   279→                </div>
   280→   280→              </div>
   281→   281→
   282→   282→              {/* Contextual actions */}
   283→   283→              <div className="flex flex-wrap items-center gap-2">
   284→   284→                {canStartReview && (
   285→   285→                  <Button disabled={busy} onClick={() => void runReviewAction({ action: "start" }, "Review started")}>
   286→   286→                    <ClipboardList className="h-4 w-4" /> Start Review
   287→   287→                  </Button>
   288→   288→                )}
   289→   289→                {st === "UNDER_REVIEW" && (
   290→   290→                  <Button
   291→   291→                    variant="outline"
   292→   292→                    onClick={() => {
   293→   293→                      setTab("submission");
   294→   294→                      toast({ title: "Add comments per section", description: "Write a comment under any section below, then submit them from the action bar." });
   295→   295→                    }}
   296→   296→                  >
   297→   297→                    <MessageSquare className="h-4 w-4" /> Add Comments
   298→   298→                  </Button>
   299→   299→                )}
   300→   300→                {canRequestRevision && (
   301→   301→                  <Button
   302→   302→                    variant="outline"
   303→   303→                    disabled={pending.length === 0 || busy}
   304→   304→                    className="border-amber-300 text-amber-800 hover:bg-amber-50 dark:text-amber-400 dark:hover:bg-amber-950/40"
   305→   305→                    onClick={() => setRevisionOpen(true)}
   306→   306→                  >
   307→   307→                    <Undo2 className="h-4 w-4" /> Request Revision
   308→   308→                  </Button>
   309→   309→                )}
   310→   310→                {canApprove &&
   311→   311→                  (hasRating ? (
   312→   312→                    <Button
   313→   313→                      disabled={busy}
   314→   314→                      className="bg-emerald-600 text-white hover:bg-emerald-700 dark:bg-emerald-500 dark:hover:bg-emerald-600"
   315→   315→                      onClick={() => setApproveOpen(true)}
   316→   316→                    >
   317→   317→                      <CheckCircle2 className="h-4 w-4" /> Approve
   318→   318→                    </Button>
   319→   319→                  ) : (
   320→   320→                    <HintButton hint="Complete the evaluation first (Evaluation tab)">
   321→   321→                      <Button disabled className="bg-emerald-600 text-white hover:bg-emerald-700 dark:bg-emerald-500">
   322→   322→                        <CheckCircle2 className="h-4 w-4" /> Approve
   323→   323→                      </Button>
   324→   324→                    </HintButton>
   325→   325→                  ))}
   326→   326→                {canFinalize && (
   327→   327→                  <Button onClick={() => void openFinalize()}>
   328→   328→                    <FileSignature className="h-4 w-4" /> Finalize Document
   329→   329→                  </Button>
   330→   330→                )}
   331→   331→                {canRegenerate && (
   332→   332→                  <Button variant="outline" onClick={() => void openFinalize()}>
   333→   333→                    <FileSignature className="h-4 w-4" /> Regenerate Final Document
   334→   334→                  </Button>
   335→   335→                )}
   336→   336→                {canArchive && (
   337→   337→                  <Button variant="ghost" disabled={busy} onClick={() => setArchiveOpen(true)}>
   338→   338→                    <Archive className="h-4 w-4" /> Archive
   339→   339→                  </Button>
   340→   340→                )}
   341→   341→              </div>
   342→   342→            </div>
   343→   343→
   344→   344→            {st === "NEEDS_REVISION" && (
   345→   345→              <Alert className="border-amber-300 bg-amber-50 text-amber-900 dark:border-amber-800 dark:bg-amber-950/40 dark:text-amber-200">
   346→   346→                <Clock className="h-4 w-4" />
   347→   347→                <AlertTitle>Waiting for barangay to resubmit</AlertTitle>
   348→   348→                <AlertDescription className="text-amber-800 dark:text-amber-300">
   349→   349→                  A revision request was sent to the barangay. You can still review the current version and approve it directly if the
   350→   350→                  submitted revision is already satisfactory.
   351→   351→                </AlertDescription>
   352→   352→              </Alert>
   353→   353→            )}
   354→   354→            {!reviewable && st !== undefined && ["NOT_STARTED", "DRAFT", "READY_FOR_SUBMISSION"].includes(st) && (
   355→   355→              <Alert>
   356→   356→                <AlertTriangle className="h-4 w-4" />
   357→   357→                <AlertTitle>Not yet submitted</AlertTitle>
   358→   358→                <AlertDescription>
   359→   359→                  This barangay has not submitted its BDRRMP yet — the content below is a read-only view of their draft.
   360→   360→                </AlertDescription>
   361→   361→              </Alert>
   362→   362→            )}
   363→   363→          </CardContent>
   364→   364→        </Card>
   365→   365→
   366→   366→        {/* Final document card */}
   367→   367→        {detail.document && <DocumentCard document={detail.document} />}
   368→   368→      </div>
   369→   369→
   370→   370→      <Tabs value={tab} onValueChange={setTab}>
   371→   371→        <TabsList>
   372→   372→          <TabsTrigger value="submission">Submission</TabsTrigger>
   373→   373→          <TabsTrigger value="evaluation">Evaluation</TabsTrigger>
   374→   374→          <TabsTrigger value="history">
   375→   375→            <History className="h-4 w-4" /> History
   376→   376→          </TabsTrigger>
   377→   377→        </TabsList>
   378→   378→
   379→   379→        {/* ---------------- Submission tab ---------------- */}
   380→   380→        <TabsContent value="submission" className="space-y-3">
   381→   381→          <Accordion type="single" collapsible defaultValue={detail.sections[0]?.key} className="space-y-3">
   382→   382→            {detail.sections.map((section, i) => (
   383→   383→              <SectionAccordion
   384→   384→                key={section.key}
   385→   385→                index={i + 1}
   386→   386→                section={section}
   387→   387→                lang={lang}
   388→   388→                values={detail.submission.values}
   389→   389→                files={filesBySection.get(section.key) ?? []}
   390→   390→                comments={commentsBySection.get(section.key) ?? []}
   391→   391→                pendingComments={pending.filter((p) => p.sectionKey === section.key)}
   392→   392→                draft={drafts[section.key] ?? ""}
   393→   393→                onDraftChange={(v) => setDrafts((d) => ({ ...d, [section.key]: v }))}
   394→   394→                onAddPending={() => addPending(section.key)}
   395→   395→                onRemovePending={(pid) => setPending((p) => p.filter((x) => x.id !== pid))}
   396→   396→                reviewable={reviewable}
   397→   397→                busy={busy}
   398→   398→              />
   399→   399→            ))}
   400→   400→          </Accordion>
   401→   401→        </TabsContent>
   402→   402→
   403→   403→        {/* ---------------- Evaluation tab ---------------- */}
   404→   404→        <TabsContent value="evaluation">
   405→   405→          <Card className="max-w-2xl">
   406→   406→            <CardHeader>
   407→   407→              <CardTitle className="text-base">MDRRMO EVALUATION</CardTitle>
   408→   408→              <CardDescription>
   409→   409→                Quality Assurance Team rating for Barangay {detail.barangay.name}&apos;s BDRRMP {detail.submission.year} (v
   410→   410→                {detail.submission.version})
   411→   411→              </CardDescription>
   412→   412→            </CardHeader>
   413→   413→            <CardContent className="space-y-4">
   414→   414→              {detail.criteria.length === 0 && <p className="text-sm text-muted-foreground">No rating criteria configured (see Settings).</p>}
   415→   415→              {detail.criteria.map((c) => (
   416→   416→                <div key={c.key} className="flex items-center gap-3">
   417→   417→                  <Label htmlFor={`score-${c.key}`} className="flex-1 text-sm font-normal">
   418→   418→                    {c.name}
   419→   419→                  </Label>
   420→   420→                  <Input
   421→   421→                    id={`score-${c.key}`}
   422→   422→                    type="number"
   423→   423→                    min={0}
   424→   424→                    max={c.maxScore}
   425→   425→                    inputMode="numeric"
   426→   426→                    className="w-20 text-right tabular-nums"
   427→   427→                    value={scores[c.key] ?? ""}
   428→   428→                    onChange={(e) => {
   429→   429→                      const v = e.target.value === "" ? undefined : Number(e.target.value);
   430→   430→                      setScores((s) => {
   431→   431→                        const next = { ...s };

   830→   830→          </span>
   831→   831→        </TooltipTrigger>
   832→   832→        <TooltipContent side="top">{hint}</TooltipContent>
   833→   833→      </Tooltip>
   834→   834→    </TooltipProvider>
   835→   835→  );
