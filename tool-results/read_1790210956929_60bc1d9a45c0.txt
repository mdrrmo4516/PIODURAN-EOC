   152→// ---------------------------------------------------------------------------

   155→function ConsoleUsers({ session }: { session: SessionInfo }) {
   156→  const { toast } = useToast();
   157→  const { data, loading, error, reload } = useLoad<UsersResponse>(() => api.adminUsers(), "admin-users");
   158→
   159→  const [busy, setBusy] = useState(false);
   160→  const [addOpen, setAddOpen] = useState(false);
   161→  const [editUser, setEditUser] = useState<AdminUserRow | null>(null);
   162→  const [resetUser, setResetUser] = useState<AdminUserRow | null>(null);
   163→  const [confirmToggle, setConfirmToggle] = useState<AdminUserRow | null>(null);
   164→  const [confirmDelete, setConfirmDelete] = useState<AdminUserRow | null>(null);
   165→
   166→  const users = data?.users ?? [];
   167→  const selfId = session.admin?.id;
   168→
   169→  const toggleUser = async (user: AdminUserRow) => {
   170→    setBusy(true);
   171→    try {
   172→      await api.adminUpdateUser(user.id, { active: !user.active });
   173→      toast({
   174→        title: user.active ? "User disabled" : "User enabled",
   175→        description: `${user.name} (${user.username}) ${user.active ? "can no longer sign in" : "can sign in again"}.`,
   176→      });
   177→      reload();
   178→    } catch (e) {
   179→      toast({
   180→        title: "Could not update user",
   181→        description: e instanceof Error ? e.message : "Please try again.",
   182→        variant: "destructive",
   183→      });
   184→    } finally {
   185→      setBusy(false);
   186→    }
   187→  };
   188→
   189→  const deleteUser = async (user: AdminUserRow) => {
   190→    setBusy(true);
   191→    try {
   192→      await api.adminDeleteUser(user.id);
   193→      toast({ title: "User deleted", description: `${user.name} (${user.username}) was removed.` });
   194→      setConfirmDelete(null);
   195→      reload();
   196→    } catch (e) {
   197→      toast({
   198→        title: "Could not delete user",
   199→        description: e instanceof Error ? e.message : "Please try again.",
   200→        variant: "destructive",
   201→      });
   202→    } finally {
   203→      setBusy(false);
   204→    }
   205→  };
   206→
   207→  return (
   208→    <div className="space-y-4">
   209→      <div className="flex flex-wrap items-center justify-between gap-2">
   210→        <p className="text-sm text-muted-foreground">
   211→          Console accounts and role assignments — changes are recorded in the audit trail
   212→        </p>
   213→        <Button onClick={() => setAddOpen(true)}>
   214→          <UserPlus className="h-4 w-4" /> Add User
   215→        </Button>
   216→      </div>
   217→
   218→      {/* Role / permission model reference */}
   219→      <Card>
   220→        <CardHeader className="pb-3">
   221→          <CardTitle className="text-base">Roles &amp; Permissions</CardTitle>
   222→          <CardDescription>What each console role can do in QAS33.</CardDescription>
   223→        </CardHeader>
   224→        <CardContent className="grid gap-3 p-6 pt-3 sm:grid-cols-3">
   225→          {ROLES.map((r) => (
   226→            <div key={r} className="rounded-lg border p-3">
   227→              <RoleBadge role={r} />
   228→              <p className="mt-2 text-xs leading-relaxed text-muted-foreground">{ADMIN_ROLE_META[r].description}</p>
   229→            </div>
   230→          ))}
   231→        </CardContent>
   232→      </Card>
   233→
   234→      <Card>
   235→        <CardHeader className="pb-3">
   236→          <CardTitle className="text-base">Console Accounts</CardTitle>
   237→          <CardDescription>
   238→            {loading ? "Loading accounts…" : `${users.length} account${users.length === 1 ? "" : "s"} • role changes are recorded in the audit trail`}
   239→          </CardDescription>
   240→        </CardHeader>
   241→        <CardContent className="p-0 pb-3">
   242→          {error ? (
   243→            <div className="p-4">
   244→              <ErrorAlert message={error} onRetry={reload} />
   245→            </div>
   246→          ) : loading ? (
   247→            <div className="p-4">
   248→              <TableSkeleton rows={4} cols={7} />
   249→            </div>
   250→          ) : users.length === 0 ? (
   251→            <div className="flex flex-col items-center gap-2 px-4 py-12 text-center">
   252→              <UsersIcon className="h-8 w-8 text-muted-foreground" />
   253→              <p className="text-sm text-muted-foreground">No console users found.</p>
   254→            </div>
   255→          ) : (
   256→            <div className="overflow-x-auto">
   257→              <Table className="min-w-[820px]">
   258→                <TableHeader>
   259→                  <TableRow>
   260→                    <TableHead className="pl-4">Username</TableHead>
   261→                    <TableHead>Name</TableHead>
   262→                    <TableHead>Position</TableHead>
   263→                    <TableHead>Role</TableHead>
   264→                    <TableHead>Status</TableHead>
   265→                    <TableHead>Last Login</TableHead>
   266→                    <TableHead className="pr-4 text-right">Actions</TableHead>
   267→                  </TableRow>
   268→                </TableHeader>
   269→                <TableBody>
   270→                  {users.map((u) => {
   271→                    const isSelf = u.id === selfId;
   272→                    const isDefault = u.username === "sysadmin";
   273→                    return (
   274→                      <TableRow key={u.id}>
   275→                        <TableCell className="pl-4 font-mono text-xs">{u.username}</TableCell>
   276→                        <TableCell>
   277→                          <div className="flex flex-wrap items-center gap-1.5 font-medium">
   278→                            {u.name}
   279→                            {isSelf && (
   280→                              <span className="rounded-full border border-primary/30 bg-primary/10 px-1.5 py-0.5 text-[10px] font-semibold text-primary">
   281→                                you
   282→                              </span>
   283→                            )}
   284→                            {isDefault && (
   285→                              <span className="rounded-full border border-slate-300 bg-slate-100 px-1.5 py-0.5 text-[10px] font-semibold text-slate-600 dark:border-slate-600 dark:bg-slate-800 dark:text-slate-300">
   286→                                Default
   287→                              </span>
   288→                            )}
   289→                          </div>
   290→                        </TableCell>
   291→                        <TableCell className="text-sm text-muted-foreground">{u.position ?? "—"}</TableCell>
   292→                        <TableCell>
   293→                          <RoleBadge role={u.role} />
   294→                        </TableCell>
   295→                        <TableCell>
   296→                          {u.active ? (
   297→                            <span className="inline-flex items-center gap-1 whitespace-nowrap text-xs font-medium text-emerald-700 dark:text-emerald-400">
   298→                              <ShieldCheck className="h-3.5 w-3.5" /> Active
   299→                            </span>
   300→                          ) : (
   301→                            <span className="inline-flex items-center gap-1 whitespace-nowrap text-xs font-medium text-red-600">
   302→                              <Ban className="h-3.5 w-3.5" /> Disabled
   303→                            </span>
   304→                          )}
   305→                        </TableCell>
   306→                        <TableCell className="whitespace-nowrap text-xs text-muted-foreground">
   307→                          {u.lastLoginAt ? formatDateTime(u.lastLoginAt) : "Never"}
   308→                        </TableCell>
   309→                        <TableCell className="pr-4 text-right">
   310→                          <DropdownMenu>
   311→                            <DropdownMenuTrigger asChild>
   312→                              <Button
   313→                                size="sm"
   314→                                variant="ghost"
   315→                                className="h-8 w-8 p-0"
   316→                                aria-label={`Actions for ${u.username}`}
   317→                                disabled={busy}
   318→                              >
   319→                                <MoreHorizontal className="h-4 w-4" />
   320→                              </Button>
   321→                            </DropdownMenuTrigger>
   322→                            <DropdownMenuContent align="end" className="w-52">
   323→                              <DropdownMenuItem onClick={() => setEditUser(u)}>
   324→                                <Pencil className="h-4 w-4" /> Edit User
   325→                              </DropdownMenuItem>
   326→                              <DropdownMenuItem onClick={() => setResetUser(u)}>
   327→                                <KeyRound className="h-4 w-4" /> Reset Password
   328→                              </DropdownMenuItem>
   329→                              <DropdownMenuSeparator />
   330→                              {isDefault && !isSelf ? (
   331→                                <>
   332→                                  <ProtectedAction
   333→                                    icon={u.active ? Ban : CheckCircle2}
   334→                                    label={u.active ? "Disable User" : "Enable User"}
   335→                                    tooltip="The default System Administrator account is protected"
   336→                                  />
   337→                                  <ProtectedAction
   338→                                    icon={Trash2}
   339→                                    label="Delete User"
   340→                                    tooltip="The default System Administrator account is protected"
   341→                                  />
   342→                                </>
   343→                              ) : isSelf ? null : (
   344→                                <>
   345→                                  <DropdownMenuItem onClick={() => setConfirmToggle(u)}>
   346→                                    {u.active ? (
   347→                                      <Ban className="h-4 w-4" />
   348→                                    ) : (
   349→                                      <CheckCircle2 className="h-4 w-4" />
   350→                                    )}{" "}
   351→                                    {u.active ? "Disable User" : "Enable User"}
   352→                                  </DropdownMenuItem>
   353→                                  <DropdownMenuItem
   354→                                    className="text-destructive focus:text-destructive"
   355→                                    onClick={() => setConfirmDelete(u)}
   356→                                  >
   357→                                    <Trash2 className="h-4 w-4" /> Delete User
   358→                                  </DropdownMenuItem>
   359→                                </>
   360→                              )}
   361→                            </DropdownMenuContent>
   362→                          </DropdownMenu>
   363→                        </TableCell>
   364→                      </TableRow>
   365→                    );
   366→                  })}
   367→                </TableBody>
   368→              </Table>
   369→            </div>
   370→          )}
   371→        </CardContent>
   372→      </Card>
   373→
   374→      {/* ---- add user ---- */}
   375→      {addOpen && (
   376→        <AddUserDialog
   377→          onClose={() => setAddOpen(false)}
   378→          onCreated={() => {
   379→            setAddOpen(false);
   380→            reload();
   381→          }}
   382→        />
   383→      )}
   384→
   385→      {/* ---- edit user (name, position, role, active) ---- */}
   386→      {editUser && (
   387→        <EditUserDialog
   388→          user={editUser}
   389→          onClose={() => setEditUser(null)}
   390→          onSaved={() => {
   391→            setEditUser(null);
   392→            reload();
   393→          }}
   394→        />
   395→      )}
   396→
   397→      {/* ---- reset password ---- */}
   398→      {resetUser && (
   399→        <ResetPasswordDialog
   400→          user={resetUser}
   401→          onClose={() => setResetUser(null)}
   402→          onDone={() => {
   403→            setResetUser(null);
   404→            reload();
   405→          }}
   406→        />
   407→      )}
   408→
   409→      {/* ---- disable / enable confirm ---- */}
   410→      <AlertDialog open={!!confirmToggle} onOpenChange={(open) => !open && !busy && setConfirmToggle(null)}>
   411→        <AlertDialogContent>
   412→          <AlertDialogHeader>
   413→            <AlertDialogTitle>{confirmToggle?.active ? "Disable this user?" : "Enable this user?"}</AlertDialogTitle>
   414→            <AlertDialogDescription>
   415→              {confirmToggle?.active
   416→                ? `${confirmToggle.name} (${confirmToggle.username}) will lose access to the MDRRMO console until re-enabled.`
   417→                : `${confirmToggle?.name} (${confirmToggle?.username}) will regain access to the MDRRMO console.`}
   418→            </AlertDialogDescription>
   419→          </AlertDialogHeader>
   420→          <AlertDialogFooter>
   421→            <AlertDialogCancel disabled={busy}>Cancel</AlertDialogCancel>
   422→            <AlertDialogAction
   423→              disabled={busy}
   424→              onClick={() => {
   425→                const user = confirmToggle;
   426→                setConfirmToggle(null);
   427→                if (user) void toggleUser(user);
   428→              }}
   429→            >
   430→              {busy && <Loader2 className="h-4 w-4 animate-spin" />}
   431→              {confirmToggle?.active ? "Disable User" : "Enable User"}
   432→            </AlertDialogAction>
   433→          </AlertDialogFooter>
   434→        </AlertDialogContent>
   435→      </AlertDialog>
   436→
   437→      {/* ---- delete confirm ---- */}
   438→      <AlertDialog open={!!confirmDelete} onOpenChange={(open) => !open && !busy && setConfirmDelete(null)}>
   439→        <AlertDialogContent>
   440→          <AlertDialogHeader>
   441→            <AlertDialogTitle className="flex items-center gap-2">
   442→              <Trash2 className="h-5 w-5 text-destructive" /> Delete this user?
   443→            </AlertDialogTitle>
   444→            <AlertDialogDescription>
   445→              {confirmDelete?.name} ({confirmDelete?.username}) will be permanently removed. This action cannot be
   446→              undone. The default System Administrator account and your own account cannot be deleted.
   447→            </AlertDialogDescription>
   448→          </AlertDialogHeader>
   449→          <AlertDialogFooter>
   450→            <AlertDialogCancel disabled={busy}>Cancel</AlertDialogCancel>
   451→            <AlertDialogAction
   452→              className="bg-destructive text-white hover:bg-destructive/90"
   453→              disabled={busy}
   454→              onClick={() => {
   455→                const user = confirmDelete;
   456→                if (user) void deleteUser(user);
   457→              }}
   458→            >
   459→              {busy && <Loader2 className="h-4 w-4 animate-spin" />}
   460→              Delete User
   461→            </AlertDialogAction>
   462→          </AlertDialogFooter>
   463→        </AlertDialogContent>
   464→      </AlertDialog>
   465→    </div>
   466→  );
   467→}

   470→// Add user dialog (mounted only while open so the form resets on close)

   472→function AddUserDialog({ onClose, onCreated }: { onClose: () => void; onCreated: () => void }) {
   473→  const { toast } = useToast();
   474→  const [username, setUsername] = useState("");
   475→  const [name, setName] = useState("");
   476→  const [position, setPosition] = useState("");
   477→  const [password, setPassword] = useState("");
   478→  const [role, setRole] = useState<AdminRole>("MDRRMO_STAFF");
   479→  const [saving, setSaving] = useState(false);
   480→
   481→  const valid = username.trim().length >= 3 && name.trim().length > 0 && password.length >= 8;
   482→
   483→  const submit = async () => {
   484→    if (!valid) return;
   485→    setSaving(true);
   486→    try {
   487→      await api.adminCreateUser({
   488→        username: username.trim(),
   489→        name: name.trim(),
   490→        position: position.trim() || undefined,
   491→        password,
   492→        role,
   493→      });
   494→      toast({
   495→        title: "User created",
   496→        description: `${name.trim()} (${username.trim()}) can now sign in as ${ADMIN_ROLE_META[role].label}.`,
   497→      });
   498→      onCreated();
   499→    } catch (e) {
   500→      toast({
   501→        title: "Could not create user",
   502→        description: e instanceof Error ? e.message : "Please try again.",
   503→        variant: "destructive",
   504→      });
   505→      setSaving(false);
   506→    }
   507→  };
   508→
   509→  return (
   510→    <Dialog open onOpenChange={(o) => !saving && !o && onClose()}>
   511→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-md">
   512→        <DialogHeader>
   513→          <DialogTitle className="flex items-center gap-2">
   514→            <UserPlus className="h-5 w-5 text-primary" /> Add Console User
   515→          </DialogTitle>
   516→          <DialogDescription>
   517→            Create an account for the MDRRMO console. Username must be 3+ characters, password 8+.
   518→          </DialogDescription>
   519→        </DialogHeader>
   520→        <div className="grid gap-3">
   521→          <div className="space-y-1.5">
   522→            <Label htmlFor="nu-username">Username</Label>
   523→            <Input
   524→              id="nu-username"
   525→              value={username}
   526→              onChange={(e) => setUsername(e.target.value)}
   527→              placeholder="e.g., mdrrmo2"
   528→              autoComplete="off"
   529→              className="font-mono"
   530→            />
   531→          </div>
   532→          <div className="space-y-1.5">
   533→            <Label htmlFor="nu-name">Full name</Label>
   534→            <Input id="nu-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g., Maria D. Santos" />
   535→          </div>
   536→          <div className="space-y-1.5">
   537→            <Label htmlFor="nu-position">Position</Label>
   538→            <Input
   539→              id="nu-position"
   540→              value={position}
   541→              onChange={(e) => setPosition(e.target.value)}
   542→              placeholder="e.g., DRRM Staff"
   543→            />
   544→          </div>
   545→          <div className="space-y-1.5">
   546→            <Label htmlFor="nu-password">Password</Label>
   547→            <Input
   548→              id="nu-password"
   549→              type="password"
   550→              value={password}
   551→              onChange={(e) => setPassword(e.target.value)}
   552→              autoComplete="new-password"
   553→            />
   554→            <p className="text-[11px] text-muted-foreground">Minimum 8 characters.</p>
   555→          </div>
   556→          <div className="space-y-1.5">
   557→            <Label htmlFor="nu-role">Role</Label>
   558→            <Select value={role} onValueChange={(v) => setRole(normalizeAdminRole(v))}>
   559→              <SelectTrigger id="nu-role">
   560→                <SelectValue />
